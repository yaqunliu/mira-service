# 英语单词视频生成功能 - 详细实现文档

## 一、需求概述

### 1.1 核心功能
- 输入3-5个英语单词，生成英语单词教学视频
- 视频结构：单词展示(5s) → 单词解释(3-5s) → 循环
- 音频由视频生成API通过提示词控制（不需要单独TTS）
- 支持API触发和聊天触发两种方式

### 1.2 配置项

| 配置项 | 类型 | 默认值 | 来源 | 说明 |
|--------|------|--------|------|------|
| words | List[str] | **必填** | 可配置1 | 3-5个英语单词 |
| word_count | int | 从words推断 | 可配置1 | 单词数量 |
| voice_gender | "female" / "male" | "female" | 可配置2 | 声音性别 |
| voice_age | "child" / "adult" | "child" | 可配置2 | 声音年龄 |
| sentence_level | "kindergarten" / "primary" / "middle" | "primary" | 可配置3 | 句子难度 |
| word_repeat_count | int | 2 | 系统默认 | 单词重复次数 |
| translation_repeat_count | int | 1 | 系统默认 | 翻译重复次数 |
| enable_sentence_video | bool | false | 系统默认 | 是否生成句子视频 |
| style | str | "cartoon" | 系统默认 | 视频风格 |

```python
from typing import List, Literal, TypedDict, Optional
from dataclasses import dataclass, field

class WordVideoConfig(TypedDict, total=False):
    """英语单词视频配置"""
    # 单词相关
    words: List[str]                                    # 单词列表（必填）
    word_count: int                                     # 单词数量
    
    # 音频相关
    word_repeat_count: int                             # 单词重复次数，默认2
    translation_repeat_count: int                      # 翻译重复次数，默认1
    
    # 句子视频
    enable_sentence_video: bool                        # 是否开启句子视频，默认False
    
    # 声音配置
    voice_gender: Literal["female", "male"]           # 声音性别，默认female
    voice_age: Literal["child", "adult"]             # 声音年龄，默认child
    
    # 难度配置
    sentence_level: Literal["kindergarten", "primary", "middle"]  # 句子难度，默认primary
    
    # 风格
    style: str                                         # 视频风格，默认cartoon


@dataclass
class WordVideoConfigDefaults:
    """默认配置值"""
    word_repeat_count: int = 2
    translation_repeat_count: int = 1
    enable_sentence_video: bool = False
    voice_gender: Literal["female", "male"] = "female"
    voice_age: Literal["child", "adult"] = "child"
    sentence_level: Literal["kindergarten", "primary", "middle"] = "primary"
    style: str = "cartoon"
    
    def to_dict(self) -> dict:
        return {
            "word_repeat_count": self.word_repeat_count,
            "translation_repeat_count": self.translation_repeat_count,
            "enable_sentence_video": self.enable_sentence_video,
            "voice_gender": self.voice_gender,
            "voice_age": self.voice_age,
            "sentence_level": self.sentence_level,
            "style": self.style,
        }


def merge_word_video_config(user_config: Optional[dict]) -> WordVideoConfig:
    """
    合并用户配置和默认值
    
    Args:
        user_config: 用户提供的配置（可以为None或空字典）
        
    Returns:
        合并后的完整配置
    """
    defaults = WordVideoConfigDefaults()
    result = defaults.to_dict()
    
    if user_config:
        result.update(user_config)
    
    # 从words推断word_count
    if "words" in result and result["words"]:
        result["word_count"] = len(result["words"])
    
    return result


def validate_word_video_config(config: WordVideoConfig) -> tuple[bool, str]:
    """
    验证配置有效性
    
    Returns:
        (is_valid, error_message)
    """
    # 验证单词
    words = config.get("words", [])
    if not words:
        return False, "单词列表不能为空"
    
    if len(words) < 3 or len(words) > 5:
        return False, "单词数量必须是3-5个"
    
    # 验证单词格式
    for word in words:
        if not word or not word.strip():
            return False, "单词不能为空"
        if not word.replace(" ", "").isalpha():
            return False, f"单词 '{word}' 包含非法字符"
    
    # 验证重复次数
    word_repeat = config.get("word_repeat_count", 2)
    if word_repeat < 1 or word_repeat > 5:
        return False, "单词重复次数必须在1-5之间"
    
    trans_repeat = config.get("translation_repeat_count", 1)
    if trans_repeat < 1 or trans_repeat > 3:
        return False, "翻译重复次数必须在1-3之间"
    
    return True, ""
```

```python
def _extract_word_video_config(user_message: str, details: dict) -> dict:
    """从用户消息和意图详情中提取单词视频配置"""
    config = {}
    
    # 从details中提取
    if "words" in details:
        config["words"] = details["words"]
    
    if "word_count" in details:
        config["word_count"] = details["word_count"]
    
    # 提取其他配置
    config_mappings = {
        "word_repeat_count": ["word_repeat_count", "word_repeat"],
        "translation_repeat_count": ["translation_repeat_count", "trans_repeat"],
        "enable_sentence_video": ["enable_sentence_video", "sentence_video", "enable_sentence"],
        "voice_gender": ["voice_gender", "gender", "女声", "男声"],
        "voice_age": ["voice_age", "age", "童声", "成人"],
        "sentence_level": ["sentence_level", "难度", "level"],
    }
    
    message_lower = user_message.lower()
    
    # 解析 voice_gender
    if "女声" in user_message:
        config["voice_gender"] = "female"
    elif "男声" in user_message:
        config["voice_gender"] = "male"
    
    # 解析 voice_age
    if "童声" in user_message or "儿童" in user_message:
        config["voice_age"] = "child"
    elif "成人" in user_message:
        config["voice_age"] = "adult"
    
    # 解析 sentence_level
    if "幼儿园" in user_message:
        config["sentence_level"] = "kindergarten"
    elif "小学" in user_message:
        config["sentence_level"] = "primary"
    elif "中学" in user_message:
        config["sentence_level"] = "middle"

    import re
    repeat_match = re.search(r"单词[念说]((\d+)次|\((\d+)\))", user_message)
    if repeat_match:
        count = repeat_match.group(1) or repeat_match.group(2)
        if count:
            config["word_repeat_count"] = int(count)
    
    # 解析句子视频
    if "句子视频" in user_message or "生成句子" in user_message:
        config["enable_sentence_video"] = True
    
    return config
```

```python
from typing import Dict, Any, List
from app.agent.workers.base_worker import BaseWorker, WorkerRegistry
from app.agent.state.schemas import ComicDramaState, ProductionStage
from app.agent.config.word_video_config import merge_word_video_config
from app.core.logger import logger

class WordVideoWorker(BaseWorker):
    """英语单词视频Worker"""
    
    @property
    def worker_type(self) -> str:
        return "word_video"
    
    async def execute(self, state: ComicDramaState) -> Dict[str, Any]:
        """执行单词视频生成"""
        logger.info("[WordVideoWorker] 开始执行")
        
        # 1. 获取并合并配置
        raw_config = state.get("word_video_config", {})
        config = merge_word_video_config(raw_config)
        
        logger.info(f"[WordVideoWorker] 配置: {config}")
        
        words = config.get("words", [])
        
        # 2. 生成所有分镜
        all_shots = []
        
        for word in words:
            # 单词展示分镜
            display_shot = await self._create_word_display_shot(word, config)
            all_shots.append(display_shot)
            
            # 单词解释分镜
            explanation_shot = await self._create_word_explanation_shot(word, config)
            all_shots.append(explanation_shot)
            
            # 可选：句子分镜
            if config.get("enable_sentence_video", False):
                sentence_shot = await self._create_sentence_shot(word, config)
                all_shots.append(sentence_shot)
        
        # 3. 提交视频生成
        await self._generate_videos(all_shots, config)
        
        # 4. 剪辑导出
        final_video_url = await self._export_video(all_shots, config)
        
        return {
            "final_video_url": final_video_url,
            "shots": all_shots,
            "current_stage": ProductionStage.COMPLETED,
        }
    
    async def _create_word_display_shot(self, word: str, config: Dict) -> Dict:
        """创建单词展示分镜"""
        word_repeat = config.get("word_repeat_count", 2)
        trans_repeat = config.get("translation_repeat_count", 1)
        
        # 获取翻译（这里可以调用翻译API）
        translation = await self._translate_word(word)
        
        # 构建音频文本
        audio_parts = [word] * word_repeat + [translation] * trans_repeat
        audio_text = ", ".join(audio_parts)
        
        # 构建视频提示词
        video_prompt = f"""
Create a vibrant, colorful abstract background with swirling colors blending together.
The background should be colorful gradient, cartoon style, dreamy and brilliant.

Style requirements:
- Multiple colors naturally blending and swirling (red, orange, yellow, green, blue, purple)
- Center of the image displays the white English word: "{word}"
- Word uses bold sans-serif font, modern and clean
- Below the word shows the Chinese translation: "{translation}"
- Overall style: Childlike, cartoon, colorful, educational

Audio requirement:
- Pronounce: "{audio_text}"
- {config.get('voice_gender', 'female')} voice, {config.get('voice_age', 'child')} tone
- Clear, child-friendly pronunciation
"""
        
        # 保存分镜
        shot = await self._save_shot({
            "type": "word_display",
            "shot_subtype": "display",
            "word": word,
            "translation": translation,
            "video_prompt": video_prompt,
            "audio_text": audio_text,
            "duration": 5,
        })
        
        return shot
    
    async def _create_word_explanation_shot(self, word: str, config: Dict) -> Dict:
        """创建单词解释分镜"""
        sentence_level = config.get("sentence_level", "primary")
        
        # 生成简单句子
        sentence = await self._generate_simple_sentence(word, sentence_level)
        
        # 选择场景
        scene = self._select_scene(sentence)
        
        video_prompt = f"""
Generate a {scene} scene in cartoon style.

Scene elements:
- Top of the frame displays the sentence: "{sentence}"
- The sentence uses white font with rounded rectangle background
- Characters naturally demonstrate the sentence meaning
- Bright colors, childlike style
- Educational and warm atmosphere
- Disney/Pixar animation quality

Audio requirement:
- Read the sentence: "{sentence}"
- {config.get('voice_gender', 'female')} voice, {config.get('voice_age', 'child')} tone
- Clear, natural pronunciation
"""
        
        shot = await self._save_shot({
            "type": "word_explanation",
            "shot_subtype": "explanation",
            "word": word,
            "sentence": sentence,
            "sentence_level": sentence_level,
            "video_prompt": video_prompt,
            "audio_text": sentence,
            "duration": 4,
        })
        
        return shot
    
    async def _create_sentence_shot(self, word: str, config: Dict) -> Dict:
        """创建句子展示分镜"""
        sentence_level = config.get("sentence_level", "primary")
        
        # 生成更复杂的句子
        sentence = await self._generate_context_sentence(word, sentence_level)
        
        video_prompt = f"""
Generate a scene showing the sentence in context.

Scene elements:
- Large sentence displayed at top: "{sentence}"
- White rounded rectangle background for text
- Characters interacting in a meaningful scene
- Cartoon style, bright colors
- Educational content

Audio requirement:
- Read the sentence naturally: "{sentence}"
- {config.get('voice_gender', 'female')} voice, {config.get('voice_age', 'child')} tone
"""
        
        shot = await self._save_shot({
            "type": "sentence_display",
            "shot_subtype": "sentence",
            "word": word,
            "sentence": sentence,
            "video_prompt": video_prompt,
            "audio_text": sentence,
            "duration": 4,
        })
        
        return shot
    
    async def _translate_word(self, word: str) -> str:
        """翻译单词（简化版，实际应该调用翻译API）"""
        # 简单词典
        translations = {
            "apple": "苹果", "banana": "香蕉", "orange": "橙子",
            "cat": "猫", "dog": "狗", "bird": "鸟",
            "book": "书", "pen": "钢笔", "chair": "椅子",
        }
        return translations.get(word.lower(), f"【{word}】")
    
    async def _generate_simple_sentence(self, word: str, level: str) -> str:
        """生成简单句子"""
        templates = {
            "kindergarten": [
                "I see a {word}.",
                "This is a {word}.",
                "A {word}.",
            ],
            "primary": [
                "I have a {word}.",
                "This is my {word}.",
                "I like a {word}.",
            ],
            "middle": [
                "I eat an {word} every day.",
                "The {word} is on the table.",
                "I want a {word}.",
            ] if word[-1] not in ['a', 'e', 'i', 'o', 'u'] else [
                "I eat {word} every day.",
                "The {word} is on the table.",
                "I want {word}.",
            ]
        }
        
        import random
        template = random.choice(templates.get(level, templates["primary"]))
        return template.format(word=word)
    
    async def _generate_context_sentence(self, word: str, level: str) -> str:
        """生成上下文句子"""
        templates = {
            "kindergarten": "I see a {word}. It is nice.",
            "primary": "I have a {word}. It is my favorite.",
            "middle": "The {word} is very important to me. I use it every day.",
        }
```

```python
        config = action_data  # 用户在卡片中选择的配置
        
        # 再次验证
        is_valid, error_msg = _validate_word_video_config(config)
        if not is_valid:
            return {
                "needs_clarification": True,
                "clarification_message": error_msg
            }
        
        # 验证通过，设置配置并开始生成
        return {
            "word_video_config": config,
            "current_stage": "WORD_VIDEO_READY",
            "needs_clarification": False
        }
    
    return {}
```

## 十三、空正文兼容设计

### 13.1 问题背景

现有漫剧Agent的流程是：
```
上传剧本(script_text) → 分析剧本 → 生成角色/场景 → 生成分镜 → ...
```

现在英语单词视频：
- 不需要剧本(script_text可以为空)
- 核心输入是配置(config)
- 直接生成单词展示分镜

### 13.2 兼容方案设计

**核心思路：让 `script_text` 变为可选字段**

#### 方案一：通过 creation_type 区分（推荐）

根据 `creation_type` 选择不同的处理逻辑：

```python
async def stage_router_node(state: ComicDramaState) -> Dict[str, Any]:
    """阶段路由节点"""
    
    creation_type = state.get("creation_type", "comic_drama")
    
    if creation_type == "word_video":
        # 单词视频：不需要剧本，直接进入分镜生成
        return await _handle_word_video_flow(state)
    
    elif creation_type == "comic_drama":
        # 漫剧：需要剧本，按原有流程
        return await _handle_comic_drama_flow(state)
```

#### 方案二：脚本为空时跳过分析

